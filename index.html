1. Collegamento Diretto Google Drive (Sincronizzazione Automatica Live)

Attiva

Incolla l'URL della tua Google Apps Script Web App per collegare il tuo account Google Drive. Ogni volta che qualcuno compila o modifica il modulo, i dati verranno inviati e salvati automaticamente su Google Fogli in background!

Come collegare il tuo account Google Drive:
  1. Apri un nuovo Foglio Google sul tuo account Google Drive.
  2. Nel menu in alto, fai clic su Estensioni > Apps Script.
  3. Incolla il codice sottostante e fai clic su Esegui Distribuzione > Nuova Distribuzione.
  4. Seleziona tipo: App Web, Imposta Accesso: Chiunque (Anyone) e clicca Distribuisci.
  5. Copia l'URL fornito da Google ed incollalo nel campo Webhook sopra!
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = JSON.parse(e.postData.contents);
  
  sheet.clearContents();
  sheet.appendRow(["Nome Partecipante", "Contatto", "Tipologia Stanza", "Stanza Assegnata", "Intolleranze/Diete", "SPA", "Garage", "Trasporto", "Rimborso (€)", "Totale Extra (€)"]);
  
  if (data.participants && Array.isArray(data.participants)) {
    data.participants.forEach(function(p) {
      sheet.appendRow([
        p.name,
        p.contact || '',
        p.roomType === 'single' ? 'Singola (+€50)' : 'Doppia (In Budget)',
        p.assignedRoom || 'Non Assegnata',
        p.dietary || 'Nessuna',
        p.spa ? 'SI (€30)' : 'NO',
        p.garage ? 'SI (€56)' : 'NO',
        p.transport,
        p.transportCost || 0,
        p.totalExtra
      ]);
    });
  }

  return ContentService.createTextOutput(JSON.stringify({"result": "success"}))
    .setMimeType(ContentService.MimeType.JSON);
}
